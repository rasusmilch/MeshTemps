// micro_cli.h - Header-only, tiny serial console router for Arduino/ESP32
// -----------------------------------------------------------------------------
// Goals
//  - Collect bytes from HardwareSerial (or any Stream), build lines, route
//    commands to handlers.
//  - Quote-aware tokenizer with C-style escapes (\" and \\\).
//  - Flat command table ("command"), plus optional subcommands
//    ("command sub ...").
//  - Typed argument helpers and automatic help generation.
//  - No dynamic allocation in steady-state other than Arduino String growth.
//  - Google style naming.
//
// Usage (summary):
//   MicroCli cli(Serial);
//   cli.Register("debug", "debug on|off", [&](Cli& c, const Tokens& t){ ... });
//   cli.Poll();  // call in loop()
//
// Integrations for MeshTemps are provided below this header (search for
// "// --- MeshTemps integration stubs ---").
// -----------------------------------------------------------------------------

#ifndef MICRO_CLI_H_
#define MICRO_CLI_H_

#include <Arduino.h>
#include <functional>
#include <vector>

namespace microcli {

struct Tokens {
  String original_line;      // Full original line (trimmed) for ROL.
  std::vector<String> parts; // Tokenized parts (command + args).
};

class Cli; // fwd

using CommandHandler = std::function<void(Cli &cli, const Tokens &tokens)>;

struct CommandSpec {
  const char *name;       // First token (case sensitive for speed; use lower).
  const char *usage;      // Short usage tail for `help`.
  CommandHandler handler; // Handler invoked when this command matches.
};

class Cli {
public:
  explicit Cli(Stream &io, size_t max_line_len = 256)
      : io_(io), max_line_len_(max_line_len) {}

  // Registers a top-level command. Name should be a single token (no spaces).
  void Register(const char *name, const char *usage, CommandHandler handler) {
    commands_.push_back({name, usage, std::move(handler)});
  }

  // Optional: override default prompt and ok/error tokens.
  void SetOkToken(const char *ok) { ok_token_ = ok; }
  void SetErrToken(const char *err) { err_token_ = err; }

  // Poll input; call in loop(). Returns true if a line was executed.
  bool Poll() {
    bool executed = false;
    while (io_.available() > 0) {
      const char ch = static_cast<char>(io_.read());
      if (ch == '\r')
        continue; // ignore CR
      if (ch != '\n') {
        if (line_buffer_.length() >= max_line_len_) {
          // Trim head if needed to keep most-recent tail.
          const int excess = static_cast<int>(line_buffer_.length()) -
                             static_cast<int>(max_line_len_ - 1);
          line_buffer_.remove(0, excess);
        }
        line_buffer_ += ch;
        continue;
      }

      // We have a line.
      line_buffer_.trim();
      if (line_buffer_.length() == 0) {
        PrintOk();
        executed = true;
        line_buffer_.clear();
        continue;
      }

      Tokens tokens;
      tokens.original_line = line_buffer_;
      Tokenize(line_buffer_, &tokens.parts);

      if (!tokens.parts.empty()) {
        executed = true;
        Execute(tokens);
      } else {
        PrintOk();
      }
      line_buffer_.clear();
    }
    return executed;
  }

  // Typed arg helpers -------------------------------------------------------
  static bool AsInt(const Tokens &t, size_t index, long *out) {
    if (index >= t.parts.size())
      return false;
    char *end = nullptr;
    long v = strtol(t.parts[index].c_str(), &end, 10);
    if (end == t.parts[index].c_str())
      return false;
    *out = v;
    return true;
  }
  static bool AsUInt(const Tokens &t, size_t index, unsigned long *out) {
    if (index >= t.parts.size())
      return false;
    char *end = nullptr;
    unsigned long v = strtoul(t.parts[index].c_str(), &end, 10);
    if (end == t.parts[index].c_str())
      return false;
    *out = v;
    return true;
  }
  static bool AsFloat(const Tokens &t, size_t index, float *out) {
    if (index >= t.parts.size())
      return false;
    char *end = nullptr;
    float v = strtof(t.parts[index].c_str(), &end);
    if (end == t.parts[index].c_str())
      return false;
    *out = v;
    return true;
  }
  static bool AsString(const Tokens &t, size_t index, String *out) {
    if (index >= t.parts.size())
      return false;
    *out = t.parts[index];
    return true;
  }

  // Returns the remainder of the original line after token `from_index`.
  static String RestOfLine(const Tokens &t, size_t from_index) {
    if (from_index >= t.parts.size())
      return String();
    // Re-tokenize to find spans; cheaper: rebuild from tokens with spaces.
    String out;
    for (size_t i = from_index; i < t.parts.size(); ++i) {
      if (i > from_index)
        out += ' ';
      out += t.parts[i];
    }
    return out;
  }

  // Prints one-line error + unified "ERR" token.
  void PrintErr(const char *msg) { io_.printf("%s %s\n", err_token_, msg); }
  void PrintOk() { io_.println(ok_token_); }

  // Emits `help` using registered usage strings.
  void PrintHelp() {
    io_.println(F("Commands:"));
    for (const auto &c : commands_) {
      io_.printf("  %s %s\n", c.name, c.usage ? c.usage : "");
    }
  }

private:
  static void Tokenize(const String &line, std::vector<String> *out) {
    out->clear();
    const int n = static_cast<int>(line.length());
    String current;
    current.reserve(32);

    bool in_quotes = false;
    for (int i = 0; i < n; ++i) {
      const char ch = line[i];
      if (in_quotes) {
        if (ch == '\\' && i + 1 < n) {
          const char next = line[i + 1];
          if (next == '"' || next == '\\') {
            current += next;
            ++i;
            continue;
          }
        }
        if (ch == '"') {
          in_quotes = false;
          continue;
        }
        if (static_cast<unsigned char>(ch) >= 0x20)
          current += ch;
        continue;
      }
      // not in quotes
      if (isspace(static_cast<unsigned char>(ch))) {
        if (!current.isEmpty()) {
          out->push_back(current);
          current = String();
        }
        continue;
      }
      if (ch == '"') {
        in_quotes = true;
        continue;
      }
      if (static_cast<unsigned char>(ch) >= 0x20)
        current += ch;
    }
    if (!current.isEmpty())
      out->push_back(current);
  }

  void Execute(const Tokens &tokens) {
    const String &cmd = tokens.parts[0];
    for (const auto &spec : commands_) {
      if (cmd.equals(spec.name)) {
        spec.handler(*this, tokens);
        return;
      }
    }
    io_.printf("%s unknown command: %s\n", err_token_, cmd.c_str());
  }

  Stream &io_;
  const size_t max_line_len_;
  String line_buffer_;
  const char *ok_token_ = "ok";
  const char *err_token_ = "ERR";
  std::vector<CommandSpec> commands_;
};

} // namespace microcli

#endif // MICRO_CLI_H_

// Registration (call once in setup()).
static void RegisterCommonCommands() {
  g_cli.Register("help", "- show command list",
                 [](Cli &c, const Tokens &t) { c.PrintHelp(); });

  g_cli.Register("debug", "on|off", [](Cli &c, const Tokens &t) {
    if (t.parts.size() < 2)
      return c.PrintErr("usage: debug on|off");
    g_debug_enabled = (t.parts[1] == "on");
    Serial.printf("debug=%s\n", g_debug_enabled ? "on" : "off");
  });
}

#if MESH_IS_ROOT
static void RegisterRootCommands() {
  RegisterCommonCommands();

  g_cli.Register("ls", "- list nodes and sensors", [](Cli &c, const Tokens &t) {
    // Reuse existing code paths in-place (compact printout).
    // NOTE: This duplicates current `ls` behavior.
    if (!g_last_seen["nodes"].is<JsonObject>())
      return c.PrintOk();
    JsonObject nodes = g_last_seen["nodes"].as<JsonObject>();
    for (JsonPair node_entry : nodes) {
      const String node_id = node_entry.key().c_str();
      JsonObject node_obj = node_entry.value();
      const String node_label = g_labels["nodes"][node_id] | "";
      if (node_label.length() > 0) {
        Serial.printf("node %s \"%s\":\n", node_id.c_str(), node_label.c_str());
      } else {
        Serial.printf("node %s:\n", node_id.c_str());
      }
      JsonObject sensors = node_obj["sensors"].as<JsonObject>();
      bool any_sensor = false;
      for (JsonPair sensor_entry : sensors) {
        any_sensor = true;
        const String addr = sensor_entry.key().c_str();
        JsonObject sensor_obj = sensor_entry.value();
        const float temp_c = sensor_obj["tC"] | NAN;
        const bool corrected = sensor_obj["corr"] | false;
        const String sl = g_labels["sensors"][addr] | "";
        const char *ts = isnan(temp_c) ? "NaN" : String(temp_c, 2).c_str();
        if (sl.length() > 0) {
          Serial.printf("  %s \"%s\" : %s%s\n", addr.c_str(), sl.c_str(), ts,
                        corrected ? " (corr)" : "");
        } else {
          Serial.printf("  %s : %s%s\n", addr.c_str(), ts,
                        corrected ? " (corr)" : "");
        }
      }
      if (!any_sensor)
        Serial.println(F("  (no sensors)"));
    }
    c.PrintOk();
  });

  g_cli.Register("units", "c|f", [](Cli &c, const Tokens &t) {
    if (t.parts.size() < 2)
      return c.PrintErr("usage: units c|f");
    const String &mode = t.parts[1];
    if (mode.equalsIgnoreCase("c")) {
      g_display_fahrenheit = false;
      SaveDisplayUnits();
      Serial.println(F("units=C"));
      GuiRequestRender();
    } else if (mode.equalsIgnoreCase("f")) {
      g_display_fahrenheit = true;
      SaveDisplayUnits();
      Serial.println(F("units=F"));
      GuiRequestRender();
    } else {
      c.PrintErr("use 'units c' or 'units f'");
    }
  });

  g_cli.Register(
      "highlight", "missing on|off | stale <minutes>",
      [](Cli &c, const Tokens &t) {
        if (t.parts.size() < 3)
          return c.PrintErr(
              "usage: highlight missing on|off | highlight stale <min>");
        const String &target = t.parts[1];
        if (target == "missing") {
          if (t.parts[2] == "on" || t.parts[2] == "off") {
            g_highlight_missing_nodes = (t.parts[2] == "on");
            SaveHighlightSettings();
            Serial.printf("highlight missing=%s\n",
                          g_highlight_missing_nodes ? "on" : "off");
            GuiRequestRender();
          } else {
            c.PrintErr("use on|off");
          }
        } else if (target == "stale") {
          long minutes = 0;
          if (!Cli::AsInt(t, 2, &minutes) || minutes <= 0)
            return c.PrintErr("minutes>0");
          g_stale_minutes_threshold = static_cast<uint32_t>(minutes);
          SaveHighlightSettings();
          Serial.printf("highlight stale=%ld min\n", minutes);
          GuiRequestRender();
        } else {
          c.PrintErr("unknown highlight target");
        }
      });

  g_cli.Register(
      "limits", "show | warn <low> <high> | alert <low> <high>",
      [](Cli &c, const Tokens &t) {
        auto ToC = [](float v) {
          return g_display_fahrenheit ? (v - 32.0f) / 1.8f : v;
        };
        auto FromC = [](float v) {
          return g_display_fahrenheit ? (v * 1.8f + 32.0f) : v;
        };
        const char *unit = g_display_fahrenheit ? "F" : "C";
        if (t.parts.size() >= 2 && t.parts[1] == "show") {
          Serial.printf("limits warn=%.2f..%.2f %s\n", FromC(g_warn_low_c),
                        FromC(g_warn_high_c), unit);
          Serial.printf("limits alert=%.2f..%.2f %s\n", FromC(g_alert_low_c),
                        FromC(g_alert_high_c), unit);
          return;
        }
        if (t.parts.size() == 4 &&
            (t.parts[1] == "warn" || t.parts[1] == "alert")) {
          float low_in = 0, high_in = 0;
          if (!Cli::AsFloat(t, 2, &low_in) || !Cli::AsFloat(t, 3, &high_in) ||
              high_in <= low_in) {
            return c.PrintErr("low<high required");
          }
          if (t.parts[1] == "warn") {
            g_warn_low_c = ToC(low_in);
            g_warn_high_c = ToC(high_in);
            SaveLimits();
            Serial.printf("limits warn=%.2f..%.2f %s\n", FromC(g_warn_low_c),
                          FromC(g_warn_high_c), unit);
          } else {
            g_alert_low_c = ToC(low_in);
            g_alert_high_c = ToC(high_in);
            SaveLimits();
            Serial.printf("limits alert=%.2f..%.2f %s\n", FromC(g_alert_low_c),
                          FromC(g_alert_high_c), unit);
          }
          GuiRequestRender();
          return;
        }
        c.PrintErr("usage: limits show | limits warn <low> <high> | limits "
                   "alert <low> <high>");
      });

  g_cli.Register("buzzer", "<len_ms> <warn_gap_ms> <alert_gap_ms>",
                 [](Cli &c, const Tokens &t) {
                   long len = 0, gw = -1, ga = -1;
                   if (!Cli::AsInt(t, 1, &len) || !Cli::AsInt(t, 2, &gw) ||
                       !Cli::AsInt(t, 3, &ga) || len <= 0 || gw < 0 || ga < 0) {
                     return c.PrintErr(
                         "usage: buzzer <len_ms> <warn_gap_ms> <alert_gap_ms>");
                   }
                   g_beep_len_ms = static_cast<uint32_t>(len);
                   g_beep_gap_warn_ms = static_cast<uint32_t>(gw);
                   g_beep_gap_alert_ms = static_cast<uint32_t>(ga);
                   SaveBuzzerSettings();
                   Serial.printf(
                       "buzzer len=%lu ms warn_gap=%lu ms alert_gap=%lu ms\n",
                       static_cast<unsigned long>(g_beep_len_ms),
                       static_cast<unsigned long>(g_beep_gap_warn_ms),
                       static_cast<unsigned long>(g_beep_gap_alert_ms));
                 });

  g_cli.Register("flash", "<interval_ms>|off", [](Cli &c, const Tokens &t) {
    if (t.parts.size() < 2)
      return c.PrintErr("usage: flash <interval_ms>|off");
    if (t.parts[1].equalsIgnoreCase("off")) {
      g_flash_interval_ms = 0;
      SaveFlashSettings();
      Serial.println(F("flash off"));
      GuiRequestRender();
      return;
    }
    long interval = 0;
    if (!Cli::AsInt(t, 1, &interval) || interval < 0)
      return c.PrintErr("interval>=0");
    g_flash_interval_ms = static_cast<uint32_t>(interval);
    SaveFlashSettings();
    Serial.printf("flash interval=%ld ms\n", interval);
    GuiRequestRender();
  });

  g_cli.Register("save", "- save labels", [](Cli &, const Tokens &) {
    SaveLabels();
    Serial.println(F("saved"));
  });
  g_cli.Register("load", "- load labels", [](Cli &, const Tokens &) {
    LoadLabels();
    Serial.println(F("loaded"));
    GuiRequestRender();
  });
  g_cli.Register("erase", "- erase labels", [](Cli &, const Tokens &) {
    EraseLabels();
    Serial.println(F("erased"));
    GuiRequestRender();
  });

  g_cli.Register("nodes", "clear - remove all known nodes",
                 [](Cli &c, const Tokens &t) {
                   if (t.parts.size() >= 2 && t.parts[1] == "clear") {
                     EraseKnownTopology();
                     SaveKnownTopology();
                     Serial.println(F("ok"));
                     GuiRequestRender();
                   } else {
                     c.PrintErr("usage: nodes clear");
                   }
                 });

  g_cli.Register("node", "rm <id> | <id> <location...>",
                 [](Cli &c, const Tokens &t) {
                   if (t.parts.size() >= 3 && t.parts[1] == "rm") {
                     const String id = t.parts[2];
                     JsonObject nodes = g_last_seen["nodes"].to<JsonObject>();
                     if (!nodes.containsKey(id)) {
                       Serial.println(F("not found"));
                       return;
                     }
                     nodes.remove(id);
                     SaveKnownTopology();
                     Serial.println(F("ok"));
                     GuiRequestRender();
                     return;
                   }
                   if (t.parts.size() >= 3) {
                     const String id = t.parts[1];
                     String label = Cli::RestOfLine(t, 2);
                     label.trim();
                     if (label.isEmpty())
                       return c.PrintErr("location required");
                     g_labels["nodes"][id] = label;
                     SaveLabels();
                     Serial.println(F("ok"));
                     GuiRequestRender();
                     return;
                   }
                   c.PrintErr("usage: node rm <id> | node <id> <location...>");
                 });

  g_cli.Register("name", "<addr16> <label...>", [](Cli &c, const Tokens &t) {
    if (t.parts.size() < 3)
      return c.PrintErr("usage: name <addr16> <label...>");
    const String addr = t.parts[1];
    String label = Cli::RestOfLine(t, 2);
    label.trim();
    if (addr.length() != 16 || label.isEmpty())
      return c.PrintErr("bad addr16 or empty label");
    g_labels["sensors"][addr] = label;
    SaveLabels();
    Serial.println(F("ok"));
    GuiRequestRender();
  });

  g_cli.Register("dummy", "on|off", [](Cli &c, const Tokens &t) {
    if (t.parts.size() < 2)
      return c.PrintErr("usage: dummy on|off");
    if (t.parts[1] == "on") {
      g_use_dummy_data = true;
      Serial.println(F("dummy=on"));
      GuiRequestRender();
    } else if (t.parts[1] == "off") {
      g_use_dummy_data = false;
      Serial.println(F("dummy=off"));
      GuiRequestRender();
    } else
      c.PrintErr("use on|off");
  });
}

#if !MESH_IS_ROOT
static void RegisterLeafCommands() {
  RegisterCommonCommands();
  g_cli.Register("scan", "- re-scan 1-Wire bus", [](Cli &, const Tokens &) {
    ScanSensors();
    Serial.println(F("ok"));
  });
  g_cli.Register("sendnow", "- send temps immediately",
                 [](Cli &, const Tokens &) {
                   SendTemperatures();
                   Serial.println(F("ok"));
                 });
  g_cli.Register("boilpt", "<inHg>", [](Cli &c, const Tokens &t) {
    float inHg = 0;
    if (!Cli::AsFloat(t, 1, &inHg))
      return c.PrintErr("usage: boilpt <inHg>");
    float TbC = boilingPointC_fromInHg(inHg);
    if (isnan(TbC))
      return c.PrintErr("invalid pressure");
    float TbF = TbC * 9.0f / 5.0f + 32.0f;
    Serial.printf("Boiling point at %.3f inHg: %.3f C (%.3f F)\n", (double)inHg,
                  (double)TbC, (double)TbF);
  });
  // Minimal CAL set (extend as needed):
  g_cli.Register("cal",
                 "ice|boil [addr16] | lock <actualC> | solve | list | show "
                 "[addr16] | clear <addr16> | save | load",
                 [](Cli &c, const Tokens &t) {
                   if (t.parts.size() < 2)
                     return c.PrintErr("see: cal help");
                   const String &sub = t.parts[1];
                   if (sub == "ice") {
                     if (t.parts.size() >= 3)
                       CalibrationStart(t.parts[2], kCalIce);
                     else
                       CalibrationStartAll(kCalIce);
                     Serial.println(F("ok"));
                     return;
                   }
                   if (sub == "boil") {
                     if (t.parts.size() >= 3)
                       CalibrationStart(t.parts[2], kCalBoil);
                     else
                       CalibrationStartAll(kCalBoil);
                     Serial.println(F("ok"));
                     return;
                   }
                   if (sub == "lock") {
                     float act = 0;
                     if (!Cli::AsFloat(t, 2, &act))
                       return c.PrintErr("cal lock <actualC>");
                     CalibrationLockAll(act);
                     Serial.println(F("ok"));
                     return;
                   }
                   if (sub == "solve") {
                     CalibrationSolveAndSaveAll();
                     Serial.println(F("ok"));
                     return;
                   }
                   if (sub == "list") {
                     for (const auto &e : g_cal_entries) {
                       const Coeff &cf = e.coeff;
                       Serial.printf("%s : a1=%.6f a0=%.6f\n", e.addr.c_str(),
                                     (double)cf.a1, (double)cf.a0);
                     }
                     return;
                   }
                   if (sub == "show") {
                     if (t.parts.size() >= 3) {
                       int idx = FindCalIndex(t.parts[2]);
                       if (idx >= 0) {
                         const Coeff &ccf = g_cal_entries[(size_t)idx].coeff;
                         Serial.printf("%s : a1=%.6f a0=%.6f\n",
                                       t.parts[2].c_str(), (double)ccf.a1,
                                       (double)ccf.a0);
                       } else
                         Serial.println(F("not found"));
                     } else {
                       for (const auto &e : g_cal_entries) {
                         const Coeff &cf = e.coeff;
                         Serial.printf("%s : a1=%.6f a0=%.6f\n", e.addr.c_str(),
                                       (double)cf.a1, (double)cf.a0);
                       }
                     }
                     return;
                   }
                   if (sub == "clear") {
                     if (t.parts.size() < 3)
                       return c.PrintErr("cal clear <addr16>");
                     ClearCalibration(t.parts[2]);
                     Serial.println(F("ok"));
                     return;
                   }
                   if (sub == "save") {
                     SaveAllCalibration();
                     Serial.println(F("saved"));
                     return;
                   }
                   if (sub == "load") {
                     LoadAllCalibration();
                     Serial.println(F("loaded"));
                     return;
                   }
                   c.PrintErr("unknown cal subcommand");
                 });
}
